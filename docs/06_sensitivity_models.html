8.1 Overview

This chapter defines the sensitivity analysis based on direct environmental and geographic predictors. Unlike the main PCA model, this model keeps the original environmental predictors in the mean component so that temperature, precipitation, DTR, NDVI, latitude, and distance to coast can be inspected directly.

The model includes methodological controls (nest box, marker type) and the same population, species, and phylogenetic random-effect structure as the main models. It is treated as a mean-only robustness model with a single global precision parameter (phi ~ 1).

Show code 
library(dplyr)
library(readr)
library(tibble)
library(knitr)

model_dir <- "models/sensitivity_models"

first_existing <- function(paths) {
  existing <- paths[file.exists(paths)]
  if (length(existing) == 0) {
    return(NA_character_)
  }
  existing[1]
}

summary_files <- list(
  first_month_mu = first_existing(c(file.path(model_dir, "summary_m_sensitivity_first_month_mu.txt")))
)

show_saved_summary <- function(summary_file, label) {
  if (!is.na(summary_file) && file.exists(summary_file)) {
    cat("<details><summary>", label, "</summary>\n\n```\n", sep = "")
    cat(readLines(summary_file), sep = "\n")
    cat("\n```\n</details>\n")
  }
}

m_sensitivity_first_month_mu <- brms::brm(
  n_epp_broods | trials(n_broods_sampled) ~
    tmp_cru_C_first_month_z +
    pre_cru_mm_first_month_log_z +
    dtr_cru_C_first_month_z +
    ndvi_first_month_median_30km_z +
    abs_latitude_z +
    distance_to_coast_z +
    nest_box +
    marker_type +
    (1 | population_id) +
    (1 | species_nonphylo) +
    (1 | gr(species_phylo, cov = phylo_mat_fit)),
  data = dat_fit,
  data2 = list(phylo_mat_fit = phylo_mat_fit),
  family = brms::beta_binomial(link = "logit", link_phi = "log"),
  prior = priors_mu,
  chains = 4,
  cores = 4,
  iter = 10000,
  warmup = 5000,
  control = list(adapt_delta = 0.99, max_treedepth = 15),
  seed = 1501,
  save_pars = brms::save_pars(all = TRUE)
)
