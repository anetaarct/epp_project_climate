3.1 Overview

This chapter documents the preparation of the analytical dataset used in the yearly extra-pair paternity (EPP) analyses. The workflow starts from the analysis input file, checks the response variables, restricts the data to socially monogamous species, defines population-level identifiers, and summarises the final dataset used in the Bayesian models.

## Basic variable cleaning

Geographic coordinates, response variables, environmental variables, and breeding-season variables were converted to numeric format where needed. We also created absolute latitude and a population identifier based on species identity and rounded geographic coordinates.

```{r}
#| label: basic-cleaning
#| echo: false

dat_clean <- dat_raw %>%
  mutate(
    lat = as.numeric(lat),
    long = as.numeric(long),
    lat_round = round(lat, 2),
    long_round = round(long, 2),
    abs_latitude = abs(lat),

    population_id = paste(
      scientific_name,
      lat_round,
      long_round,
      sep = "_"
    ),

    n_epp_broods = as.numeric(n_epp_broods),
    n_wpp_broods = as.numeric(n_wpp_broods),
    n_broods_sampled = as.numeric(n_broods_sampled),

    tmp_cru_C_first_month = as.numeric(tmp_cru_C_first_month),
    pre_cru_mm_first_month = as.numeric(pre_cru_mm_first_month),
    dtr_cru_C_first_month = as.numeric(dtr_cru_C_first_month),
    ndvi_first_month_median_30km = as.numeric(ndvi_first_month_median_30km),

    tmp_cru_C_full_season = as.numeric(tmp_cru_C_full_season),
    pre_cru_mm_full_season = as.numeric(pre_cru_mm_full_season),
    dtr_cru_C_full_season = as.numeric(dtr_cru_C_full_season),
    ndvi_full_season_median_30km = as.numeric(ndvi_full_season_median_30km),

    tmp_long_anom_1961_1989_C_first_month = as.numeric(tmp_long_anom_1961_1989_C_first_month),
    tmp_short_anom_5yr_C_first_month = as.numeric(tmp_short_anom_5yr_C_first_month),
    pre_long_anom_1961_1989_mm_first_month = as.numeric(pre_long_anom_1961_1989_mm_first_month),
    pre_short_anom_5yr_mm_first_month = as.numeric(pre_short_anom_5yr_mm_first_month),

    distance_to_coast_km = as.numeric(distance_to_coast_km),
    breeding_season_length_months = as.numeric(breeding_season_length_months),

    id_reference = factor(id_reference),
    population_id = factor(population_id),
    scientific_name = factor(scientific_name),
    marker_type = factor(marker_type),
    nest_box = factor(nest_box),
    breeding_system = factor(breeding_system)
  )
```

## Reference-population overlap

The overlap between reference identity (`id_reference`) and population identity (`population_id`) is checked directly from the model-ready dataset. This summary was used to evaluate whether reference identity and population identity represented sufficiently distinct grouping structures. Because population identity captures repeated observations from the same species-locality combination, it was retained as the primary repeated-observation grouping factor in the full environmental models.

```{r}
#| label: reference-population-overlap-data
#| echo: false

if (file.exists("data/processed/dat_model_phylo.rds")) {
  overlap_dat <- readRDS("data/processed/dat_model_phylo.rds")
} else if (file.exists("data/processed/dat_model.rds")) {
  overlap_dat <- readRDS("data/processed/dat_model.rds")
} else {
  overlap_dat <- read_csv("epp_data_June2026.csv", show_col_types = FALSE) %>%
    mutate(
      lat_round = round(as.numeric(lat), 2),
      long_round = round(as.numeric(long), 2),
      population_id = paste(scientific_name, lat_round, long_round, sep = "_")
    )
}

overlap_dat <- overlap_dat %>%
  mutate(
    id_reference = if ("id_reference" %in% names(.)) {
      as.character(id_reference)
    } else {
      as.character(id)
    }
  ) %>%
  mutate(
    population_id = as.character(population_id)
  ) %>%
  filter(!is.na(id_reference), !is.na(population_id))

population_overlap <- overlap_dat %>%
  group_by(population_id) %>%
  summarise(
    n_records = n(),
    n_references = n_distinct(id_reference),
    id_references = paste(sort(unique(id_reference)), collapse = "; "),
    .groups = "drop"
  )

reference_overlap <- overlap_dat %>%
  group_by(id_reference) %>%
  summarise(
    n_records = n(),
    n_populations = n_distinct(population_id),
    .groups = "drop"
  )
```

```{r}
#| label: reference-population-overlap-table
#| echo: false

population_overlap %>%
  count(n_references, name = "n_populations") %>%
  mutate(percent_populations = 100 * n_populations / sum(n_populations)) %>%
  knitr::kable(digits = 2)
```

## Environmental window correlations

The main environmental models use environmental values from the first breeding-season month. As a data check, we compared these values with the corresponding full-season summaries for temperature, precipitation, diurnal temperature range, and NDVI. Because the correlations were high, subsequent analyses use the first-month environmental values and do not treat full-season summaries as a separate predictor set.

```{r}
#| label: first-month-full-season-correlations
#| echo: false

environment_window_pairs <- tribble(
  ~variable, ~first_month, ~full_season,
  "Temperature", "tmp_cru_C_first_month", "tmp_cru_C_full_season",
  "Precipitation", "pre_cru_mm_first_month", "pre_cru_mm_full_season",
  "DTR", "dtr_cru_C_first_month", "dtr_cru_C_full_season",
  "NDVI", "ndvi_first_month_median_30km", "ndvi_full_season_median_30km"
)

environment_window_correlations <- purrr::pmap_dfr(
  environment_window_pairs,
  function(variable, first_month, full_season) {
    keep <- complete.cases(dat_model[, c(first_month, full_season)])
    test <- cor.test(
      dat_model[[first_month]][keep],
      dat_model[[full_season]][keep],
      method = "pearson"
    )

    tibble(
      variable = variable,
      n_complete = sum(keep),
      pearson_r = unname(test$estimate),
      lower_95_ci = test$conf.int[1],
      upper_95_ci = test$conf.int[2]
    )
  }
)

environment_window_correlations %>%
  mutate(across(where(is.numeric), ~ round(.x, 3))) %>%
  knitr::kable()
```
